import type { MetadataRoute } from "next";

export default function robots(): MetadataRoute.Robots {
  const base = process.env.NEXT_PUBLIC_SITE_URL ?? "https://www.donasipeduli.web.id";
  return {
    rules: {
      userAgent: "*",
      allow: "/",
      disallow: ["/admin", "/dashboard", "/login", "/register", "/reset-sandi", "/auth"],
    },
    sitemap: `${base}/sitemap.xml`,
  };
}
